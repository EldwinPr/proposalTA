\chapter{DESAIN KONSEP SOLUSI}
\label{chap:desain-konsep-solusi}

\section{Gambaran Umum Solusi}
\begin{itemize}
    \item Penelitian ini mengusulkan sebuah kerangka kerja untuk mengevaluasi kelayakan teknis dan finansial dari sistem ERP pada SBC.
    \item Metode evaluasi didasarkan pada dua pilar utama: \textit{benchmarking} performa aplikasi dan analisis \textit{Total Cost of Ownership} (TCO).
    \item Objek yang dievaluasi adalah platform SBC (studi kasus: Orange Pi 5 Pro) yang akan dibandingkan dengan platform \textit{cloud} VPS sebagai \textit{baseline}.
\end{itemize}

\section{Arsitektur Solusi}
\begin{itemize}
    \item \textbf{Arsitektur A (On-Premise SBC):} Merepresentasikan model investasi CAPEX, di mana sistem ERP diinstalasi dan dijalankan secara lokal pada perangkat Orange Pi.
    \item \textbf{Arsitektur B (Cloud VPS):} Merepresentasikan model investasi OPEX, di mana sistem ERP berjalan di server virtual yang disewa dari penyedia \textit{cloud}.
    \item Kedua arsitektur akan menggunakan tumpukan perangkat lunak (OS, web server, basis data, ERP) yang identik untuk memastikan perbandingan yang adil.
\end{itemize}

\section{Desain Implementasi}
\subsection{Penyiapan Lingkungan Perangkat Keras}
\begin{itemize}
    \item \textbf{SBC:} Melakukan instalasi sistem operasi Linux (misal: Armbian) pada Orange Pi 5 Pro, serta melakukan konfigurasi jaringan dasar.
    \item \textbf{VPS:} Melakukan provisi \textit{virtual machine} pada \textit{cloud provider} dengan spesifikasi yang memenuhi \textit{minimum requirement} ERPNext. Untuk fase penelitian, Oracle Cloud Free Tier akan digunakan untuk evaluasi performa guna mengurangi biaya operasional.
\end{itemize}

\subsection{Penyiapan Lingkungan Perangkat Lunak}
\begin{itemize}
    \item \textbf{Dependensi:} Instalasi semua komponen yang dibutuhkan, termasuk Python, Nginx, Supervisor, dan server basis data yang sesuai (PostgreSQL/MariaDB).
    \item \textbf{ERP:} Instalasi dan konfigurasi ERPNext pada kedua platform dengan dataset demo yang identik.
\end{itemize}

\section{Desain Pengujian dan Evaluasi}
\subsection{Skenario Pengujian}
\begin{itemize}
    \item Skenario pengujian dirancang untuk mensimulasikan alur kerja bisnis yang realistis pada sistem ERP \cite{maddodi2018generating}.
    \item \textbf{Operasi Baca:} Pengujian akses ke data master dan laporan (misal: membuka daftar produk, melihat riwayat penjualan).
    \item \textbf{Operasi Tulis:} Pengujian pembuatan entitas data baru (misal: membuat \textit{sales order}, validasi faktur).
    \item \textbf{Operasi Campuran:} Simulasi beban dari beberapa pengguna virtual yang melakukan operasi baca dan tulis secara bersamaan.
\end{itemize}

\subsection{Alat Benchmarking}
\begin{itemize}
    \item Simulasi beban akan dieksekusi menggunakan perangkat lunak \textit{load testing} (kandidat utama: Locust).
    \item Skenario pengujian akan ditulis sebagai kode Python (dalam Locust) untuk memastikan akurasi dan kemudahan repetisi.
\end{itemize}

\subsection{Pengumpulan Data Metrik}
\begin{itemize}
    \item \textbf{Metrik Kinerja Aplikasi:} Data \textit{throughput} (RPS) dan waktu respons (ms) akan dicatat secara otomatis oleh Locust.
    \item \textbf{Metrik Sumber Daya:} Data utilisasi CPU, RAM, dan I/O disk pada sisi server akan dipantau menggunakan utilitas `vmstat` dan `iostat`.
\end{itemize}

\section{Desain Analisis}
\subsection{Analisis Kinerja Teknis}
\begin{itemize}
    \item Membuat grafik perbandingan performa (misal: throughput vs. jumlah pengguna) antara platform SBC dan VPS.
    \item Menganalisis grafik utilisasi sumber daya untuk mengidentifikasi potensi \textit{bottleneck} pada setiap platform.
    \item Mengevaluasi stabilitas dan tingkat kesalahan (\textit{error rate}) pada kedua platform di bawah beban puncak.
\end{itemize}

\subsection{Analisis Biaya (TCO)}
\begin{itemize}
    \item Menghitung dan membandingkan TCO untuk periode 3 dan 5 tahun.
    \item \textbf{Komponen Biaya SBC:} Meliputi biaya perangkat keras (CAPEX) dan estimasi biaya konsumsi listrik (OPEX).
    \item \textbf{Komponen Biaya VPS Komersial:} Meliputi biaya sewa bulanan dari \textit{commercial} VPS (Hostinger, Biznet, atau DigitalOcean) yang diproyeksikan untuk periode 3 dan 5 tahun.
    \item \textbf{Catatan Metodologis:} Platform dengan biaya Rp 0 (\textit{free tier}) tidak dimasukkan dalam analisis kuantitatif TCO karena tidak menghasilkan \textit{break-even point} yang bermakna. Evaluasi performa Oracle Cloud Free Tier akan digunakan untuk \textit{qualitative trade-off discussion}.
    \item Membuat grafik perbandingan TCO untuk mengidentifikasi \textit{break-even point} antara investasi SBC dan biaya sewa VPS komersial.
\end{itemize}
